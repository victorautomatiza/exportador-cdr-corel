'==============================================================================
' CONFIGURAï¿½ï¿½ES LEMBRADAS ENTRE USOS (registro do Windows, via GetSetting)
'==============================================================================
Private Sub CarregarConfiguracao()
    txtOrigem.Text = GetSetting(CHAVE_REG, "Config", "Origem", "")
    txtDestino.Text = GetSetting(CHAVE_REG, "Config", "Destino", "")

    Select Case Val(GetSetting(CHAVE_REG, "Config", "Formato", CStr(FMT_PDF)))
        Case FMT_JPEG: optJPEG.Value = True
        Case FMT_PNG: optPNG.Value = True
        Case Else: optPDF.Value = True
    End Select

    chkPaginas.Value = (GetSetting(CHAVE_REG, "Config", "Paginas", "0") = "1")
    chkCurvas.Value = (GetSetting(CHAVE_REG, "Config", "Curvas", "0") = "1")
    chkGrupos.Value = (GetSetting(CHAVE_REG, "Config", "Grupos", "0") = "1")

    If GetSetting(CHAVE_REG, "Config", "Modo", "P") = "A" Then
        optModoAberto.Value = True
    Else
        optModoPasta.Value = True
    End If
    AtualizarControlesModo
End Sub

Private Sub SalvarConfiguracao()
    On Error Resume Next
    SaveSetting CHAVE_REG, "Config", "Origem", Trim$(txtOrigem.Text)
    SaveSetting CHAVE_REG, "Config", "Destino", Trim$(txtDestino.Text)
    SaveSetting CHAVE_REG, "Config", "Formato", CStr(FormatoSelecionado())
    SaveSetting CHAVE_REG, "Config", "Paginas", IIf(chkPaginas.Value, "1", "0")
    SaveSetting CHAVE_REG, "Config", "Curvas", IIf(chkCurvas.Value, "1", "0")
    SaveSetting CHAVE_REG, "Config", "Grupos", IIf(chkGrupos.Value, "1", "0")
    SaveSetting CHAVE_REG, "Config", "Modo", IIf(optModoAberto.Value, "A", "P")
End Sub

Private Function FormatoSelecionado() As Long
    If optJPEG.Value Then
        FormatoSelecionado = FMT_JPEG
    ElseIf optPNG.Value Then
        FormatoSelecionado = FMT_PNG
    Else
        FormatoSelecionado = FMT_PDF
    End If
End Function

Private Sub btnOrigem_Click()
    Dim p As String
    p = EscolherPasta("Selecione a pasta que contï¿½m os arquivos .CDR")
    If Len(p) > 0 Then txtOrigem.Text = p
End Sub

Private Sub btnDestino_Click()
    Dim p As String
    p = EscolherPasta("Selecione a pasta onde os arquivos exportados serï¿½o salvos")
    If Len(p) > 0 Then txtDestino.Text = p
End Sub

Private Sub btnSalvar_Click()
    Dim fso As Object
    Dim origem As String, destino As String
    Dim res As ResultadoExportacao
    Dim ok As Boolean

    ' Modo "documento aberto": fluxo separado (nao usa origem/destino configurados)
    If optModoAberto.Value Then
        ExportarDocumentoAbertoClick
        Exit Sub
    End If

    Set fso = CreateObject("Scripting.FileSystemObject")
    origem = Trim$(txtOrigem.Text)
    destino = Trim$(txtDestino.Text)

    ' --- Validaï¿½ï¿½es ---
    If Len(origem) = 0 Or Not fso.FolderExists(origem) Then
        MsgBox "Selecione uma pasta de origem vï¿½lida.", vbExclamation, "Exportador de CDR"
        txtOrigem.SetFocus
        Exit Sub
    End If
    If Len(destino) = 0 Then
        MsgBox "Selecione a pasta de destino.", vbExclamation, "Exportador de CDR"
        txtDestino.SetFocus
        Exit Sub
    End If
    If StrComp(fso.GetAbsolutePathName(origem), fso.GetAbsolutePathName(destino), vbTextCompare) = 0 Then
        MsgBox "A pasta de destino precisa ser diferente da pasta de origem.", _
               vbExclamation, "Exportador de CDR"
        txtDestino.SetFocus
        Exit Sub
    End If

    SalvarConfiguracao

    ' --- Executa ---
    DefinirOcupado True
    ok = ProcessarPasta(origem, destino, FormatoSelecionado(), _
                        CBool(chkPaginas.Value), CBool(chkGrupos.Value), _
                        CBool(chkCurvas.Value), Me, res)
    DefinirOcupado False

    If ok Then
        MostrarRelatorio res, destino
    Else
        lblStatus.Caption = "Pronto para exportar."
        lblBarra.Width = 0
    End If
End Sub

' Exporta o documento aberto: a pasta de destino e SEMPRE escolhida aqui
Private Sub ExportarDocumentoAbertoClick()
    Dim destino As String
    Dim res As ResultadoExportacao
    Dim ok As Boolean

    If Not VerificarDocumentoAberto() Then Exit Sub

    destino = EscolherPasta("Selecione a pasta de destino para exportar o documento aberto")
    If Len(destino) = 0 Then
        MsgBox "A pasta de destino ï¿½ obrigatï¿½ria para exportar o documento aberto." & vbCrLf & _
               "Nenhum arquivo foi exportado.", vbInformation, "Exportador de CDR"
        Exit Sub
    End If

    SalvarConfiguracao

    DefinirOcupado True
    ok = ExportarDocumentoAberto(destino, FormatoSelecionado(), CBool(chkPaginas.Value), _
                                 CBool(chkGrupos.Value), CBool(chkCurvas.Value), Me, res)
    DefinirOcupado False

    If ok Then
        MostrarRelatorio res, destino
    Else
        lblStatus.Caption = "Pronto para exportar."
        lblBarra.Width = 0
    End If
End Sub

Private Sub optModoPasta_Click()
    AtualizarControlesModo
End Sub

Private Sub optModoAberto_Click()
    AtualizarControlesModo
End Sub

' Habilita/desabilita origem e destino conforme o modo e ajusta os textos
Private Sub AtualizarControlesModo()
    Dim pasta As Boolean, livre As Boolean

    pasta = optModoPasta.Value
    livre = Not mProcessando

    txtOrigem.Enabled = livre And pasta
    btnOrigem.Enabled = livre And pasta
    txtDestino.Enabled = livre And pasta
    btnDestino.Enabled = livre And pasta

    If pasta Then
        lblOrigem.Caption = "Arquivos (pasta de origem dos .CDR)"
        lblDestino.Caption = "Destino (pasta onde serï¿½o salvos os exportados)"
        btnSalvar.Caption = "Salvar"
    Else
        lblOrigem.Caption = "Origem: documento aberto no CorelDRAW (pasta de origem nï¿½o ï¿½ usada)"
        lblDestino.Caption = "Destino: a pasta serï¿½ solicitada ao clicar em Exportar"
        btnSalvar.Caption = "Exportar"
    End If
End Sub

Private Sub btnSair_Click()
    If mProcessando Then
        ' Durante o processamento o botï¿½o vira "Cancelar": termina o arquivo atual e para
        mCancelar = True
        btnSair.Enabled = False
        lblStatus.Caption = "Cancelando apï¿½s o arquivo atual..."
    Else
        Unload Me
    End If
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If mProcessando Then
        Cancel = True               ' nï¿½o deixa fechar no meio do processamento
        mCancelar = True
        lblStatus.Caption = "Cancelando apï¿½s o arquivo atual..."
    End If
End Sub


'==============================================================================
' ESTADO "OCUPADO" E PROGRESSO (chamados pelo mï¿½dulo durante o processamento)
'==============================================================================
Private Sub DefinirOcupado(ByVal ocupado As Boolean)
    mProcessando = ocupado
    If ocupado Then mCancelar = False

    fraModo.Enabled = Not ocupado
    fraFormato.Enabled = Not ocupado
    chkPaginas.Enabled = Not ocupado
    chkGrupos.Enabled = Not ocupado
    chkCurvas.Enabled = Not ocupado
    btnSalvar.Enabled = Not ocupado
    AtualizarControlesModo          ' origem/destino dependem do modo
    btnSair.Enabled = True

    If ocupado Then
        btnSair.Caption = "Cancelar"
        lblBarra.Width = 0
        lblStatus.Caption = "Preparando..."
    Else
        btnSair.Caption = "Sair"
    End If
    Me.Repaint
End Sub

Public Sub AtualizarProgresso(ByVal atual As Long, ByVal total As Long, ByVal texto As String)
    lblStatus.Caption = texto
    If total > 0 Then lblBarra.Width = lblBarraFundo.Width * atual / total
    Me.Repaint
    DoEvents
End Sub

Public Function CancelamentoSolicitado() As Boolean
    DoEvents
    CancelamentoSolicitado = mCancelar
End Function


'==============================================================================
' SELETOR DE PASTA (Shell.Application - funciona em qualquer versï¿½o do Corel)
'==============================================================================
Private Function EscolherPasta(ByVal titulo As String) As String
    Dim sh As Object, pasta As Object
    Dim h As Long

    On Error Resume Next
    h = Application.AppWindow.Handle       ' mantï¿½m o diï¿½logo na frente do Corel
    Set sh = CreateObject("Shell.Application")
    Set pasta = sh.BrowseForFolder(h, titulo, &H41)    ' somente pastas + estilo novo
    If Not pasta Is Nothing Then EscolherPasta = pasta.Self.Path
End Function
